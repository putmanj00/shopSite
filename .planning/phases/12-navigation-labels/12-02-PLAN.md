**New state and refs (add to existing useState/useEffect block):**
```typescript
const [shopOpen, setShopOpen] = useState(false);
const dropdownRef = useRef<HTMLDivElement>(null);

// Close dropdown on Escape key or outside click
useEffect(() => {
  const handleKey = (e: KeyboardEvent) => {
    if (e.key === 'Escape') setShopOpen(false);
  };
  const handleClickOutside = (e: MouseEvent) => {
    if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
      setShopOpen(false);
    }
  };
  document.addEventListener('keydown', handleKey);
  document.addEventListener('mousedown', handleClickOutside);
  return () => {
    document.removeEventListener('keydown', handleKey);
    document.removeEventListener('mousedown', handleClickOutside);
  };
}, []);
```

**Replace the entire nav links section** (the `<div className="hidden lg:flex items-center gap-6 lg:gap-8">` block containing the flat links) with:

